```python
class FrontierExplorer(Node):
    def __init__(self) -> None:
        super().__init__("frontier_explorer")

        params = [
            ("map_topic", "/map"),
            ("map_frame", "map"),
            ("base_frame", "base_link"),
            ("navigate_action", "/navigate_to_pose"),
            ("exploration_period", 1.0),
            ("goal_timeout", 35.0),
            ("frontier_cooldown", 20.0),
            ("min_frontier_size", 3),
            ("min_goal_distance", 0.35),
            ("max_goal_distance", 3.0),
            ("free_threshold", 20),
        ]
        self.declare_parameters(namespace="", parameters=cast(Any, params))

        self.latest_map: OccupancyGrid | None = None
        self.current_goal_key: str | None = None
        self.current_goal_sent_seconds: float | None = None
        self.goal_handle: Any | None = None
        self.cooled_frontiers: dict[str, float] = {}
```

这些字段分别保存最新地图、当前 action goal、goal 发送时间、goal handle 和冷却中的 frontier key。

## 第四步：保存地图并实现探索 tick

回到 class 缩进级别，继续追加：

```python

    def _map_callback(self, msg: OccupancyGrid) -> None:
        self.latest_map = msg

    def _exploration_tick(self) -> None:
        if self.current_goal_key is not None:
            self._cancel_timed_out_goal()
            return

        if self.latest_map is None:
            self.get_logger().debug("waiting for /map before selecting frontier")
            return

        if not self.action_client.wait_for_server(timeout_sec=0.1):
            self.get_logger().debug("waiting for NavigateToPose action server")
            return

        robot_pose = self._lookup_robot_pose()
        if robot_pose is None:
            return

        candidate = self._select_candidate(robot_pose[0], robot_pose[1])
        if candidate is None:
            self.get_logger().info("no usable frontier candidate found")
            return

        goal = NavigateToPose.Goal()
        goal.pose.header.frame_id = cast(str, self.get_parameter("map_frame").value)
        goal.pose.header.stamp = self.get_clock().now().to_msg()
        goal.pose.pose.position.x = candidate.map_x
        goal.pose.pose.position.y = candidate.map_y
        goal.pose.pose.orientation.w = 1.0

        self.current_goal_key = candidate.key
        self.current_goal_sent_seconds = self._now_seconds()
        self.get_logger().info(
            "sending frontier %s at (%.2f, %.2f), size=%d, distance=%.2f, score=%.2f"
            % (candidate.key, candidate.map_x, candidate.map_y, candidate.size, candidate.distance, candidate.score)
        )
        future = self.action_client.send_goal_async(goal)
        future.add_done_callback(self._goal_response_callback)
```

这里的 goal 使用 `map` frame，坐标来自上一节的 free-side centroid。

## 第五步：查询 TF 并选择候选

继续追加：

```python

    def _lookup_robot_pose(self) -> tuple[float, float] | None:
        map_frame = cast(str, self.get_parameter("map_frame").value)
        base_frame = cast(str, self.get_parameter("base_frame").value)
        try:
            transform = self.tf_buffer.lookup_transform(map_frame, base_frame, rclpy.time.Time())
        except TransformException as exc:
            self.get_logger().debug(f"waiting for {map_frame} -> {base_frame}: {exc}")
            return None
        return (
            transform.transform.translation.x,
            transform.transform.translation.y,
        )

    def _select_candidate(self, robot_x: float, robot_y: float):
        assert self.latest_map is not None
        grid = self.latest_map.info
        info = GridInfo(
            width=grid.width,
            height=grid.height,
            resolution=grid.resolution,
            origin_x=grid.origin.position.x,
            origin_y=grid.origin.position.y,
        )
        candidates = find_frontier_candidates(
            self.latest_map.data,
            info,
            robot_x,
            robot_y,
            min_frontier_size=cast(int, self.get_parameter("min_frontier_size").value),
            min_goal_distance=cast(float, self.get_parameter("min_goal_distance").value),
            max_goal_distance=cast(float, self.get_parameter("max_goal_distance").value),
            free_threshold=cast(int, self.get_parameter("free_threshold").value),
        )
        available = filter_cooldown_candidates(candidates, self.cooled_frontiers, self._now_seconds())
        return available[0] if available else None
```

不要回退到 `/odom`。探索目标和机器人位置都使用 `map` 坐标。

## 第六步：处理 action response 和 result

继续追加：

```python

    def _goal_response_callback(self, future: Any) -> None:
        goal_handle = future.result()
        if not goal_handle.accepted:
            self.get_logger().warning(f"frontier goal {self.current_goal_key} was rejected")
            self._cool_current_frontier()
            self._clear_current_goal()
            return

        self.goal_handle = goal_handle
        result_future = goal_handle.get_result_async()
        result_future.add_done_callback(self._goal_result_callback)

    def _goal_result_callback(self, future: Any) -> None:
        result = future.result()
        status = result.status
        if status == GoalStatus.STATUS_SUCCEEDED:
            self.get_logger().info(f"frontier goal {self.current_goal_key} succeeded")
        else:
            self.get_logger().warning(f"frontier goal {self.current_goal_key} finished with status {status}")
            self._cool_current_frontier()
        self._clear_current_goal()
```

rejected 和非 succeeded 结果都会冷却当前 frontier；succeeded 只清空当前 goal。

## 第七步：实现超时、冷却、清理和时间

继续追加：

```python

    def _cancel_timed_out_goal(self) -> None:
        if self.goal_handle is None or self.current_goal_sent_seconds is None:
            return

        timeout = cast(float, self.get_parameter("goal_timeout").value)
        if self._now_seconds() - self.current_goal_sent_seconds <= timeout:
            return

        self.get_logger().warning(f"frontier goal {self.current_goal_key} timed out; canceling")
        self.goal_handle.cancel_goal_async()
        self._cool_current_frontier()
        self._clear_current_goal()

    def _cool_current_frontier(self) -> None:
        if self.current_goal_key is None:
            return
        cooldown = cast(float, self.get_parameter("frontier_cooldown").value)
        self.cooled_frontiers[self.current_goal_key] = self._now_seconds() + cooldown

    def _clear_current_goal(self) -> None:
        self.current_goal_key = None
        self.current_goal_sent_seconds = None
        self.goal_handle = None

    def _now_seconds(self) -> float:
        return self.get_clock().now().nanoseconds / 1e9
```

超时路径会 cancel goal，并把 frontier 放入冷却。

## 第八步：追加 main 入口

最后追加：

```python


def main(args=None) -> None:
    explorer = None
    try:
        rclpy.init(args=args)
        explorer = FrontierExplorer()
        rclpy.spin(explorer)
    except (KeyboardInterrupt, ExternalShutdownException):
        pass
    finally:
        if explorer is not None:
            explorer.destroy_node()
        if rclpy.ok():
            rclpy.shutdown()


if __name__ == "__main__":
    main()
```
